{{- /* Liquid's truncatewords: the first n words, and "..." when there were more. dict "s" "n" */ -}} {{- $w := split (strings.TrimSpace .s) " " | complement (slice "") -}} {{- $out := delimit (first .n $w) " " -}} {{- if gt (len $w) .n }}{{ $out = printf "%s..." $out }}{{ end -}} {{- return $out -}}
